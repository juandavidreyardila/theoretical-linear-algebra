\begin{definition}[Linear Functional]
    A linear functional on $V$ is a linear map from $V$ to $F$; that is,
    an element of $\mathcal{L}(V, F)$.
\end{definition}


\begin{definition}[Dual Space]
    The dual space of $V$, denoted by $V'$, is the vector space of all linear 
    functionals on $V$. In other words $V' = \mathcal{L}(V, F)$. 
\end{definition}


\begin{theorem}
    Suppose $V$ is finite-dimensional. Then $V'$ is also finite-dimensional and
    \begin{equation*}
        \dim V' = \dim V.
    \end{equation*}
\end{theorem}

\begin{proof}
    $\dim \mathcal{L}(V, F) = (\dim V)(\dim F) = \dim V$.
\end{proof}
